& {
    Set-StrictMode -Version Latest
    $ErrorActionPreference = 'Stop'
    $script = 'C:\Users\user\Desktop\SmartFactory\read-quarantine-pre-v1020.ps1'
    $expected = 'EBEB041A42D0FC2232ADCDDA13547C8151209C22E8D85F1B1B1D70D7021A1765'
    $stream = $null
    $sha = $null
    try {
        $native = [IO.Path]::Combine(
            [Environment]::SystemDirectory,
            'WindowsPowerShell\v1.0\powershell.exe'
        )
        foreach ($path in @($script, ($script + '.sha256.txt'))) {
            $node = [IO.FileInfo]::new($path)
            while ($null -ne $node) {
                if (([IO.File]::GetAttributes($node.FullName) -band
                    [IO.FileAttributes]::ReparsePoint) -ne 0) {
                    throw 'Reparse transfer path rejected.'
                }
                if ($node -is [IO.FileInfo]) { $node = $node.Directory }
                else { $node = $node.Parent }
            }
        }
        $lines = @([IO.File]::ReadAllLines($script + '.sha256.txt'))
        if ($lines.Count -ne 1 -or $lines[0] -cne $expected) {
            throw 'Helper SHA256 sidecar mismatch.'
        }
        $stream = [IO.File]::Open($script, 'Open', 'Read', 'Read')
        $sha = [Security.Cryptography.SHA256]::Create()
        $hash = [BitConverter]::ToString($sha.ComputeHash($stream)).Replace('-', '')
        if ($stream.Length -ne 24114 -or $hash -cne $expected) {
            throw 'Helper size or SHA256 mismatch.'
        }
        Write-Host '[LAUNCH] Verified read-only quarantine recheck. No deletion or restart.'
        & $native -NoLogo -NoProfile -ExecutionPolicy Bypass -File $script
        if ($LASTEXITCODE -ne 0) { throw "Read-only recheck stopped: exit=$LASTEXITCODE" }
    }
    finally {
        if ($null -ne $sha) { $sha.Dispose() }
        if ($null -ne $stream) { $stream.Dispose() }
    }
}
